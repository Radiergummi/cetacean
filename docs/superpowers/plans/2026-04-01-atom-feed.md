// internal/api/atom/atom.go
package atom

import (
	"encoding/xml"
	"io"
	"time"
)

const namespace = "http://www.w3.org/2005/Atom"

// Feed is an Atom 1.0 feed document (RFC 4287).
type Feed struct {
	XMLName xml.Name  `xml:"feed"`
	NS      string    `xml:"xmlns,attr"`
	Title   string    `xml:"title"`
	ID      string    `xml:"id"`
	Updated time.Time `xml:"updated"`
	Links   []Link    `xml:"link"`
	Entries []Entry   `xml:"entry"`
}

// ContentElement renders <content type="text">...</content> per RFC 4287.
type ContentElement struct {
	Type  string `xml:"type,attr"`
	Value string `xml:",chardata"`
}

// Entry is a single Atom feed entry.
type Entry struct {
	ID         string         `xml:"id"`
	Title      string         `xml:"title"`
	Updated    time.Time      `xml:"updated"`
	Content    ContentElement `xml:"content"`
	Links      []Link         `xml:"link"`
	Categories []Category     `xml:"category"`
}

// Link is an Atom link element.
type Link struct {
	Rel  string `xml:"rel,attr"`
	Href string `xml:"href,attr"`
	Type string `xml:"type,attr,omitempty"`
}

// Category is an Atom category element.
type Category struct {
	Term string `xml:"term,attr"`
}

// Render writes the feed as Atom XML to w.
func Render(w io.Writer, f Feed) error {
	f.NS = namespace
	if _, err := io.WriteString(w, xml.Header); err != nil {
		return err
	}
	enc := xml.NewEncoder(w)
	enc.Indent("", "  ")
	return enc.Encode(f)
}
```

func (h *History) List(q HistoryQuery) []HistoryEntry {
	h.mu.RLock()
	defer h.mu.RUnlock()

	limit := q.Limit
	if limit <= 0 {
		limit = 50
	}

	if q.ResourceID != "" {
		return h.listByResource(q.ResourceID, q.Type, q.BeforeID, limit)
	}

	var result []HistoryEntry
	total := h.size
	if !h.full {
		total = h.cursor
	}

	pastCursor := q.BeforeID == 0

	for i := 0; i < total && len(result) < limit; i++ {
		idx := h.cursor - 1 - i
		if idx < 0 {
			idx += h.size
		}

		e := h.entries[idx]
		if !pastCursor {
			if e.ID == q.BeforeID {
				pastCursor = true
			}
			continue
		}

		if q.Type != "" && e.Type != q.Type {
			continue
		}

		if q.NameContains != "" && !strings.Contains(
			strings.ToLower(e.Name), strings.ToLower(q.NameContains),
		) {
			continue
		}

		result = append(result, e)
	}

	return result
}
```

import "net/http"

func contentNegotiated(
	jsonHandler, atomHandler http.HandlerFunc,
	spa http.Handler,

	return func(w http.ResponseWriter, r *http.Request) {
		switch ContentTypeFromContext(r.Context()) {
		case ContentTypeHTML:
			spa.ServeHTTP(w, r)
		case ContentTypeSSE:
			writeErrorCode(w, r, "API001", "this endpoint does not support text/event-stream")

// internal/api/atom_handlers.go
package api

import (
	"bytes"
	"fmt"
	"net/http"
	"strconv"
	"strings"
	"time"

	atomxml "github.com/radiergummi/cetacean/internal/api/atom"
	"github.com/radiergummi/cetacean/internal/auth"
	"github.com/radiergummi/cetacean/internal/cache"
)

// writeCachedAtom renders an Atom feed, computes an ETag, and returns
// 304 Not Modified if the client's If-None-Match header matches.
func writeCachedAtom(w http.ResponseWriter, r *http.Request, f atomxml.Feed) {
	rc := http.NewResponseController(w)
	_ = rc.SetWriteDeadline(time.Now().Add(30 * time.Second))

	var buf bytes.Buffer
	if err := atomxml.Render(&buf, f); err != nil {
		w.Header().Set("Cache-Control", "no-store")
		writeErrorCode(w, r, "API009", "failed to serialize response")
		return
	}

	body := buf.Bytes()
	etag := computeETag(body)
	w.Header().Set("ETag", etag)
	w.Header().Set("Content-Type", "application/atom+xml;charset=utf-8")
	w.Header().Set("Cache-Control", "no-cache")

	if etagMatch(r.Header.Get("If-None-Match"), etag) {
		w.WriteHeader(http.StatusNotModified)
		return
	}

	w.Write(body) //nolint:errcheck
}

// feedID builds a tag URI for a feed based on the request host and path.
func feedID(r *http.Request) string {
	host := r.Header.Get("X-Forwarded-Host")
	if host == "" {
		host = r.Host
	}
	return fmt.Sprintf("tag:%s,%d:%s", host, time.Now().Year(), r.URL.Path)
}

// parseAtomPagination reads cursor pagination params from the query string.
func parseAtomPagination(r *http.Request) (beforeID uint64, limit int) {
	limit = 50
	q := r.URL.Query()
	if v := q.Get("limit"); v != "" {
		if n, err := strconv.Atoi(v); err == nil && n > 0 && n <= 200 {
			limit = n
		}
	}
	if v := q.Get("before"); v != "" {
		if n, err := strconv.ParseUint(v, 10, 64); err == nil {
			beforeID = n
		}
	}
	return
}

// historyToEntries converts HistoryEntry slice to Atom entries.
func historyToEntries(r *http.Request, entries []cache.HistoryEntry) []atomxml.Entry {
	result := make([]atomxml.Entry, 0, len(entries))
	for _, e := range entries {
		content := e.Summary
		if content == "" {
			content = e.Action + " " + e.Name
		}

		title := content
		if len(title) > 80 {
			title = title[:80] + "..."
		}

		entry := atomxml.Entry{
			ID:      fmt.Sprintf("urn:cetacean:history:%d", e.ID),
			Title:   title,
			Updated: e.Timestamp,
			Content: atomxml.ContentElement{Type: "text", Value: content},
			Categories: []atomxml.Category{
				{Term: string(e.Type)},
			},
		}

		// Link to the resource detail page.
		if e.ResourceID != "" {
			entry.Links = []atomxml.Link{
				{Rel: "alternate", Href: absPath(r.Context(), resourcePath(e.Type, e.ResourceID))},
			}
		}

		result = append(result, entry)
	}
	return result
}

// resourcePath returns the URL path for a resource given its type and ID.
func resourcePath(typ cache.EventType, id string) string {
	switch typ {
	case cache.EventNode:
		return "/nodes/" + id
	case cache.EventService:
		return "/services/" + id
	case cache.EventTask:
		return "/tasks/" + id
	case cache.EventConfig:
		return "/configs/" + id
	case cache.EventSecret:
		return "/secrets/" + id
	case cache.EventNetwork:
		return "/networks/" + id
	case cache.EventVolume:
		return "/volumes/" + id
	case cache.EventStack:
		return "/stacks/" + id
	default:
		return "/"
	}
}

// paginationLinks builds self/next/previous Link elements for a feed.
func paginationLinks(r *http.Request, entries []cache.HistoryEntry, beforeID uint64, limit int) []atomxml.Link {
	basePath := absPath(r.Context(), r.URL.Path) + ".atom"
	links := []atomxml.Link{
		{Rel: "self", Href: basePath + "?" + r.URL.RawQuery},
		{Rel: "alternate", Href: absPath(r.Context(), r.URL.Path), Type: "application/json"},
	}

	// Next page: oldest entry in current result set.
	if len(entries) == limit {
		oldest := entries[len(entries)-1]
		links = append(links, atomxml.Link{
			Rel:  "next",
			Href: fmt.Sprintf("%s?before=%d&limit=%d", basePath, oldest.ID, limit),
		})
	}

	return links
}

func (h *Handlers) atomListHandler(
	title string,

	return func(w http.ResponseWriter, r *http.Request) {

	}

// HandleAtomRecommendations serves recommendations as an Atom feed.
func (h *Handlers) HandleAtomRecommendations(w http.ResponseWriter, r *http.Request) {
	if !h.requireAnyGrant(w, r) {
		return
	}
	results := h.recEngine.Results()
	lastTick := h.recEngine.LastTick()
	if lastTick.IsZero() {
		lastTick = time.Now()
	}

	entries := make([]atomxml.Entry, 0, len(results))
	for _, rec := range results {
		entries = append(entries, atomxml.Entry{
			ID:      fmt.Sprintf("urn:cetacean:recommendation:%s:%s", rec.TargetID, rec.Category),
			Title:   rec.Message,
			Updated: lastTick,
			Content: atomxml.ContentElement{Type: "text", Value: rec.Message},
			Links: []atomxml.Link{
				{Rel: "alternate", Href: absPath(r.Context(), "/recommendations")},
			},
			Categories: []atomxml.Category{
				{Term: string(rec.Category)},
				{Term: string(rec.Severity)},
			},
		})
	}

	feed := atomxml.Feed{
		Title:   "Cetacean — Recommendations",
		ID:      feedID(r),
		Updated: lastTick,
		Links: []atomxml.Link{
			{Rel: "self", Href: absPath(r.Context(), r.URL.Path) + ".atom"},
			{Rel: "alternate", Href: absPath(r.Context(), "/recommendations"), Type: "application/json"},
		},
		Entries: entries,
	}
	writeCachedAtom(w, r, feed)
}

// filterHistoryACL filters history entries by ACL read permission.
func (h *Handlers) filterHistoryACL(r *http.Request, entries []cache.HistoryEntry) []cache.HistoryEntry {
	if h.acl == nil {
		return entries
	}
	id := auth.IdentityFromContext(r.Context())
	result := entries[:0:0]
	for _, e := range entries {
		resource := string(e.Type) + ":" + e.Name
		if h.acl.Can(id, "read", resource) {
			result = append(result, e)
		}
	}
	return result
}
